function requireGuest(req: Request, res: Response, next: NextFunction): void {
  if (!req.session.guest) {
    req.session.returnTo = req.originalUrl;

    res.redirect("/search?reason=search");
    return;
  }

  next();
}

app.post("/search/access-code", (req: Request, res: Response) => {
  try {
    const accessCode = String(req.body.accessCode ?? "");

    if (!accessCode.trim) {
      return res.status(400).json({
        success: false,
        key: "nullAccessCode",
      });
    }

    const guest = req.session.pendingGuest!;

    if (!guest || guest.accessCode! !== accessCode) {
      return res.status(400).json({
        success: false,
        key: "unmatchedAccessCode",
      });
    }


    if (guest && guest.accessCode === accessCode) {
      req.session.guest = guest;
      delete req.session.pendingGuest;

      const redirectUrl = req.session.returnTo || "/";

      console.debug("redirectUrl: " + redirectUrl);

      delete req.session.returnTo;

        //const navButtons = getNavButtons(req);
        //const StepperState = getNewStep(req);

      return res.status(200).json({
        success: true,
        redirectUrl,
       // navButtons,
        //stepper: getStepper(req),
        //steps: STEPS,
      });
    }
  }
  catch (error) {
    console.error("Access code verification failed:", error);

    return res.status(500).render("search", {
      error: "We were unable to verify your information.",
    });
  }
});

app.post("/api/stepper", requireGuest, (req, res) => {
  const { currentStep, completedSteps } = req.body;

  const validSteps: StepId[] = [
    "welcome",
    "party",
    "star",
    "rsvp",
    "thats60s",
    "thats7s",
    "end",
  ];

  if (
    !validSteps.includes(currentStep) ||
    !Array.isArray(completedSteps) ||
    !completedSteps.every((step) => validSteps.includes(step))
  ) {
    return res.status(400).json({
      success: false,
      message: "Invalid stepper data",
    });
  }

  setCurrentStep(req, currentStep);

  for (const step of completedSteps) {
    completeStep(req, step);
  }

  return res.json({
    success: true,
    stepper: getStepper(req),
  });
});

function guestHasRsvp(req: Request): boolean {
  const guest = req.session.guest ?? null;

  if (guest == null) {
    return false;
  }

  const partySizeEst = Number(guest?.partySizeEst ?? -1);

  return partySizeEst >= 0;
}

function guestHasStar(req: Request): boolean {
  const guest = req.session.guest ?? null;

  if (guest == null) {
    return false;
  }

  const star = typeof guest?.star === "string" ? guest?.star : "";

  return star.trim() !== "";
}

function guestHasSongs(req: Request): boolean {
  const guest = req.session.guest ?? null;

  if (guest == null) {
    return false;
  }

  const hasSongs =
    (typeof guest?.song1 === "string" && guest?.song1.trim() !== "") ||
    (typeof guest?.song2 === "string" && guest?.song2.trim() !== "") ||
    (typeof guest?.song3 === "string" && guest?.song3.trim() !== "") ||
    (typeof guest?.song4 === "string" && guest?.song4.trim() !== "") ||
    (typeof guest?.song5 === "string" && guest?.song5.trim() !== "") ||
    (typeof guest?.song6 === "string" && guest?.song6.trim() !== "") ||
    (typeof guest?.song7 === "string" && guest?.song7.trim() !== "");

  return hasSongs;
}
  
function guestIsFinished(req: Request): boolean {
  const guest = req.session.guest ?? null;

  if (guest == null) {
    return false;
  }

  const finished =
    typeof guest?.finished === "string" ? guest?.finished : "false";

  return finished === "true";
}
