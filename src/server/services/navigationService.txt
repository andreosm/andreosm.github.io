export function getStepFromPage(path: string, panel: StarPanel = 1): StepId {
  switch (path) {
    case "/":
    case "/index":
      return "welcome";
    case "/party":
      return "party";
    case "/rsvp":
      return "rsvp";
    case "/finish":
      return "end";
    case "/star":
      return panel === 1 ? "star" : panel === 2 ? "thats60s" : "thats7s";
    default:
      return "welcome";
  }
}

function button(
  visible: boolean,
  url: string | null = null,
  action: NextAction = "none",
): NavigationButton {
  return { visible, url, action };
}


export function getNavigationState(
  req: Request,
  progress: GuestProgress,
  currentStep: StepId,
): NavigationState {
  const complete = isStepsCompleted(req, progress);
  const guidedMode = !complete;
  const index = getStepIndex(currentStep);

  const previous = index > 0 ? STEPS[index - 1] : null;
  const next = index < STEPS.length - 1 ? STEPS[index + 1] : null;

  let backButton: NavigationButton;
  let nextButton: NavigationButton;

  if (guidedMode) {
    backButton = button(
      previous !== null,
      previous?.url ?? null,
      previous ? "navigate" : "none",
    );

    // RSVP and the two later Star panels save before advancing.
    const submitBeforeNext: StepId[] = ["rsvp", "thats60s", "thats7s"];

    nextButton = button(
      next !== null,
      next?.url ?? null,
      !next
        ? "none"
        : submitBeforeNext.includes(currentStep)
          ? "submit-navigate"
          : "navigate",
    );
  } else {
    backButton = button(false);

    switch (currentStep) {
      case "rsvp":
      case "thats7s":
        // Save and show feedback without leaving the page.
        nextButton = button(true, null, "submit");
        break;

      case "star":
        nextButton = button(true, "/star?panel=2", "navigate");
        break;

      case "thats60s":
        nextButton = button(true, "/star?panel=3", "submit-navigate");
        break;

      default:
        nextButton = button(false);
    }
  }

  // const steps: StepView[] = STEPS.map((step) => ({
  //   ...step,
  //   active: step.id === currentStep,
  //   completed: complete || progress.completedSteps.includes(step.id),
  // }));

  // const navButtons: NavButtons = {
  //   showNext: nextButton.visible,
  //   showNextLink: nextButton.action === "navigate",
  //   showBack: backButton.visible,
  //   showRsvp: complete,
  //   showStar: complete,
  //   showLogout: progress.identified,
  //   nextUrl: nextButton.url,
  //   backUrl: backButton.url,
  // };

  return {
    showNavbar: complete,
    guidedMode,
    back: backButton,
    next: nextButton,
    // navButtons,
    // steps,
    // stepper: {
    //   currentStep,
    //   completedSteps: steps
    //     .filter((step) => step.completed)
    //     .map((step) => step.id),
    // },
  };
}

export function getNextIncompleteStep(
  progress: GuestProgress,
): StepDefinition | null {
  return (
    STEPS.find((step) => !progress.completedSteps.includes(step.id)) ?? null
  );
}
